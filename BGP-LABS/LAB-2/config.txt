R1
----
en
conf t
hostname R1
int e1/0
duplex full
no sh
ip add 10.0.0.1 255.255.255.252
int e1/1
duplex full
no sh
ip add 10.0.0.5 255.255.255.252
int e1/3
duplex full
no sh
ip add 150.1.1.1 255.255.255.0

int loop0
ip add 1.1.1.1 255.255.255.255
no sh
exit

router eigrp 254
network 10.0.0.0 0.0.0.3
network 10.0.0.4 0.0.0.3
network 1.1.1.1 0.0.0.0

router bgp 254
bgp router-id 1.1.1.1
neighbor 2.2.2.2 remote-as 254
neighbor 2.2.2.2 password 0 CCNP
neighbor 2.2.2.2 update-source loop0
neighbor 2.2.2.2 route-reflector-client

neighbor 3.3.3.3 remote-as 254
neighbor 3.3.3.3 password 0 CCNP
neighbor 3.3.3.3 update-source loop0

neighbor 4.4.4.4 remote-as 254
neighbor 4.4.4.4 password 0 CCNP
neighbor 4.4.4.4 update-source loop0
neighbor 4.4.4.4 route-reflector-client

bgp cluster-id 13.13.13.13

timers bgp 5 15

neighbor 3.3.3.3 send-community
neighbor 2.2.2.2 send-community

redistribute connected route-map REDISTRIBUTE

exit

route-map REDISTRIBUTE
match interface e1/3
set community no-export
